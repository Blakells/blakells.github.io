# blakells.dev Portfolio Site Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build Blake Bailey's portfolio site (blakells.dev) as specified in
`docs/superpowers/specs/2026-11-10-portfolio-site-design.md`, with every check in the spec's section 7
running locally and in CI, ready for the gated launch steps.

**Architecture:** A static Astro 7 site. Content lives in Markdown/MDX collections and two YAML data

- [ ] **Step 4: Create `scripts/lib/serve.mjs`**

```js
import { createServer } from 'node:http';
import { readFile, stat } from 'node:fs/promises';
import { extname, join, normalize, resolve, sep } from 'node:path';

const TYPES = {
  '.html': 'text/html; charset=utf-8',
  '.css': 'text/css; charset=utf-8',
  '.js': 'text/javascript; charset=utf-8',
  '.xml': 'application/xml; charset=utf-8',
  '.json': 'application/json; charset=utf-8',
  '.txt': 'text/plain; charset=utf-8',
  '.svg': 'image/svg+xml',
  '.png': 'image/png',
  '.jpg': 'image/jpeg',
  '.webp': 'image/webp',
  '.ico': 'image/x-icon',
  '.pdf': 'application/pdf',
  '.webmanifest': 'application/manifest+json',
};

/**
 * Serve a built site directory over HTTP on 127.0.0.1, the way GitHub Pages does:
 * directories serve their index.html, and a missing path gets 404.html with status 404.
 * @param {string} dir
 * @param {number} [port] 0 picks a free port
 * @returns {Promise<{ url: string, close: () => Promise<void> }>}
 */
export async function serveDir(dir, port = 0) {
  const root = resolve(dir);
  const server = createServer(async (req, res) => {
    const path = decodeURIComponent(new URL(req.url ?? '/', 'http://localhost').pathname);
    let file = normalize(join(root, path));
    if (file !== root && !file.startsWith(root + sep)) {
      res.writeHead(403).end();
      return;
    }
    try {
      if ((await stat(file)).isDirectory()) file = join(file, 'index.html');
      const body = await readFile(file);
      res.writeHead(200, { 'content-type': TYPES[extname(file)] ?? 'application/octet-stream' }).end(body);
    } catch {
      const page = await readFile(join(root, '404.html')).catch(() => Buffer.from('Not found'));
      res.writeHead(404, { 'content-type': 'text/html; charset=utf-8' }).end(page);
    }
  });
  await new Promise((ready) => server.listen(port, '127.0.0.1', () => ready(undefined)));
  const address = server.address();
  const actual = typeof address === 'object' && address ? address.port : port;
  return {
    url: `http://127.0.0.1:${actual}`,
    close: () => new Promise((done) => server.close(() => done(undefined))),
  };
}
```

```ts
import { parse } from 'yaml';
import type { z } from 'astro/zod';

/** Parse a YAML string and validate it, naming the file and the failing fields in the error. */
export function parseYamlData<T extends z.ZodType>(raw: string, schema: T, file: string): z.infer<T> {
  const result = schema.safeParse(parse(raw));
  if (!result.success) {
    const fields = result.error.issues.map((issue) => `${issue.path.join('.') || '(root)'}: ${issue.message}`);
    throw new Error(`${file} is invalid:\n${fields.join('\n')}`);
  }
  return result.data;
}
```

- [ ] **Step 3: Create `src/lib/facts.ts`**

```ts
import { site } from './site';

const FACTS: Record<string, string> = {
  preflightTests: String(site.preflight.tests),
  preflightCi: site.preflight.ci,
  inferenceServer: site.inferenceServer,
  degreeStatus: site.degree.status,
};

const TOKEN = /\{([a-zA-Z]+)\}/g;

/** Replace {token} placeholders with facts from site.yaml. Unknown camelCase tokens throw. */
export function fillFacts(text: string): string {
  return text.replace(TOKEN, (whole, name: string) => {
    const value = FACTS[name];
    if (value === undefined) throw new Error(`Unknown fact token ${whole}`);
    return value;
  });
}

/** fillFacts applied to every string inside a value (objects and arrays are copied). */
export function fillFactsDeep<T>(value: T): T {
  if (typeof value === 'string') return fillFacts(value) as T;
  if (Array.isArray(value)) return value.map((item) => fillFactsDeep(item)) as T;
  if (value && typeof value === 'object') {
    return Object.fromEntries(Object.entries(value).map(([key, item]) => [key, fillFactsDeep(item)])) as T;
  }
  return value;
}
```

The token pattern only matches a single camelCase word in braces, so `{not a token}` (with spaces)
is left alone.

- [ ] **Step 5: Create `tools/preflight_guard_shot.py`**

It screenshots Preflight's dry-run guard blocking a final Submit on Preflight's own test fixture page,
filled with the fictional example applicant. Nothing leaves 127.0.0.1, and it does not run `preflight apply`.

```python
"""Screenshot Preflight's dry-run guard blocking a final Submit on its own test fixture page.

Run with Preflight's virtualenv:
    python tools/preflight_guard_shot.py <preflight-repo> <out.png>
Uses only the repo's fixtures and its fictional example applicant; nothing leaves 127.0.0.1.
"""
import json
import sys
import threading
from functools import partial
from http.server import SimpleHTTPRequestHandler, ThreadingHTTPServer
from pathlib import Path
from tempfile import TemporaryDirectory

from playwright.sync_api import sync_playwright
from preflight.apply import helper_files


class QuietHandler(SimpleHTTPRequestHandler):
    def log_message(self, *args):
        pass


def main(repo: Path, out: Path) -> None:
    profile = json.loads((repo / "profile.example.json").read_text(encoding="utf-8"))
    with TemporaryDirectory() as tmp:
        helper_files.render_helpers(Path(tmp), ctx=helper_files.job_context(profile))
        guard = Path(tmp) / "dry_run_guard.js"
        handler = partial(QuietHandler, directory=str(repo / "tests" / "fixtures" / "guard"))
        server = ThreadingHTTPServer(("127.0.0.1", 0), handler)
        threading.Thread(target=server.serve_forever, daemon=True).start()
        try:
            with sync_playwright() as p:
                browser = p.chromium.launch()
                context = browser.new_context(viewport={"width": 900, "height": 260}, device_scale_factor=2)
                context.add_init_script(path=str(guard))
                page = context.new_page()
                page.goto(f"http://127.0.0.1:{server.server_address[1]}/plain-submit.html")
                page.click("#final-submit")
                page.wait_for_selector("[data-preflight-guard]")
                banner = page.locator("[data-preflight-guard]").inner_text()
                if "final Submit blocked" not in banner:
                    raise SystemExit(f"unexpected banner: {banner!r}")
                page.screenshot(path=str(out))
                browser.close()
        finally:
            server.shutdown()
    print(f"wrote {out}: {banner}")


if __name__ == "__main__":
    main(Path(sys.argv[1]), Path(sys.argv[2]))
```

````mdx
## The problem

Applying for jobs is mostly form-filling: the same facts, typed into a different applicant tracking
system every time. An AI agent can do that work. An agent that drives a real browser can also click the
final Submit when it shouldn't, answer a question with something you never said, or apply to the same job

## What it does

Preflight finds postings, filters and scores them against your profile with an LLM, tailors your resume
and cover letter to each one, renders them as PDFs, and then hands each application to a browser agent
that fills in the employer's form.

## Safety layers

The hard part is the last step. A dry run is meant to complete the whole form and stop before the final
submit, and that stop is enforced in the page, not only requested in the prompt.

A guard script runs in every page and frame before the site's own scripts. It reads the form's step
markers and blocks the final submit, whether by a click, `form.submit()` or `requestSubmit()`, while Next
and Save & Continue pass on earlier steps. A hook refuses any JavaScript the agent writes itself during a
dry run, so the agent cannot route around the guard.

The known gap, stated plainly: the guard does not see a page's own script posting the form with `fetch`.

## Anti-detection, plainly

Job sites run bot detection, and an automated browser session gets blocked or served endless CAPTCHAs.
Preflight includes measures so it can complete applications anyway:

- **The driver.** A pinned Patchright build of the Playwright MCP server attaches to Chrome without the
  `Runtime.enable` call most detectors look for.
- **Natural input.** The agent's clicks move the mouse along a short, slightly curved path, pause, and
  hold the button for a human-length moment instead of jumping to the exact center of the element.
- **CAPTCHAs.** Paid solvers run only with the operator's own key. Cloudflare's checkbox waits for a
  person to solve it in the visible window by default, and a challenge nobody can solve ends the attempt.
- **Proxy.** An optional residential proxy for CAPTCHA-related traffic, with both exit IPs checked to
  match before the agent starts.

These exist so that a person applying to jobs for themselves is not locked out. The operator is
responsible for following each site's terms; Preflight does not create fake identities and only ever
applies as you.

## Testing

The suite has {site.preflight.tests} tests and runs in under 30 seconds with no network, no API key and
no real browser profile. It uses a fake OpenAI-compatible LLM server and a fictional applicant, and it runs
the dry-run guard in headless Chromium against local pages, checking by the requests the server receives
that the final submit never arrives while the allowed steps do. CI runs the suite on Linux (Python 3.11 to
3.13) and macOS. Nothing in CI or the test suite submits an application.

I directed AI coding agents through the build with a test-first workflow, and had an independent AI
````

```js
import { readFile } from 'node:fs/promises';
import { getDocument } from 'pdfjs-dist/legacy/build/pdf.mjs';

export const RESUME_HEADINGS = ['Blake Bailey', 'Summary', 'Experience', 'Technical Projects', 'Education'];
export const RESUME_MAX_PAGES = 3;

/**
 * The text of a PDF. Runs are joined as written, with a newline where pdf.js marks a line end,
 * and NFKC folds ligature glyphs (so "ﬂ" reads as "fl").
 * @param {string} path
 * @returns {Promise<{ pages: number, text: string }>}
 */
export async function pdfText(path) {
  const data = new Uint8Array(await readFile(path));
  const doc = await getDocument({ data, useSystemFonts: true }).promise;
  const pages = [];
  for (let n = 1; n <= doc.numPages; n++) {
    const content = await (await doc.getPage(n)).getTextContent();
    pages.push(content.items.map((item) => ('str' in item ? item.str + (item.hasEOL ? '\n' : '') : '')).join(''));
  }
  return { pages: doc.numPages, text: pages.join('\n').normalize('NFKC') };
}

/**
 * Problems with a resume PDF's extracted text; an empty list means it passes.
 * @param {{ pages: number, text: string }} pdf
 * @returns {string[]}
 */
export function checkResumePdf({ pages, text }) {
  const problems = [];
  if (!text.trim()) problems.push('the PDF has no extractable text');
  if (pages > RESUME_MAX_PAGES) problems.push(`the PDF runs to ${pages} pages (limit ${RESUME_MAX_PAGES})`);
  const lower = text.toLowerCase();
  for (const heading of RESUME_HEADINGS) {
    if (!lower.includes(heading.toLowerCase())) problems.push(`the PDF text lacks "${heading}"`);
  }
  return problems;
}
```

```js
/**
 * @typedef {{ id: string, message: string, pattern: RegExp, proseOnly?: boolean, allow?: (context: string) => boolean }} Rule
 * @typedef {{ rule: string, message: string, line: number }} Violation
 */

/** @type {Rule[]} */
export const RULES = [
  { id: 'personal-email', message: 'a personal Gmail address', pattern: /@gmail\.com/gi },
  { id: 'finance-size', message: "a phrase that sizes Blake's finances", pattern: /\b(?:four|five|six|seven|eight|nine|ten)-figure\b/gi },
  { id: 'money', message: 'a dollar amount', pattern: /\$\s?\d/g, proseOnly: true },
  {
    id: 'upstream-name',
    message: '"ApplyPilot" outside the credit to the upstream project',
    pattern: /ApplyPilot/g,
    allow: (context) => context.includes('Pickle-Pixel'),
  },
  { id: 'placeholder', message: 'a placeholder', pattern: /\b(?:TODO|TBD)\b|lorem ipsum/gi },
];

/**
 * Find rule matches in a text.
 * @param {string} text
 * @param {'all' | 'general' | 'prose'} which  general: rules safe for code; prose: only prose rules; all: both
 * @returns {Violation[]}
 */
export function findViolations(text, which) {
  /** @type {Violation[]} */
  const found = [];
  for (const rule of RULES) {
    if (which === 'general' && rule.proseOnly) continue;
    if (which === 'prose' && !rule.proseOnly) continue;
    for (const match of text.matchAll(rule.pattern)) {
      const at = match.index ?? 0;
      if (rule.allow?.(text.slice(Math.max(0, at - 200), at + 200))) continue;
      found.push({ rule: rule.id, message: rule.message, line: text.slice(0, at).split('\n').length });
    }
  }
  return found;
}

/**
 * The text a visitor sees on an HTML page: scripts, styles and tags removed, common entities decoded.
 * @param {string} html
 */
export function visibleText(html) {
  return html
    .replace(/<script\b[\s\S]*?<\/script>/gi, ' ')
    .replace(/<style\b[\s\S]*?<\/style>/gi, ' ')
    .replace(/<[^>]+>/g, ' ')
    .replace(/&nbsp;/g, ' ')
    .replace(/&lt;/g, '<')
    .replace(/&gt;/g, '>')
    .replace(/&quot;/g, '"')
    .replace(/&#39;/g, "'")
    .replace(/&#36;/g, '$')
    .replace(/&amp;/g, '&');
}
```

```js
// Applies the content rules to the site's sources, the built pages and the resume PDF's text.
// Prints file, line and rule for each problem, never the matched text.
import { existsSync } from 'node:fs';
import { readdir, readFile } from 'node:fs/promises';
import { extname, join } from 'node:path';
import { findViolations, visibleText } from './lib/content-rules.mjs';
import { pdfText } from './lib/pdf-text.mjs';

const TEXT = new Set(['.astro', '.ts', '.mjs', '.js', '.css', '.md', '.mdx', '.yaml', '.yml', '.json', '.svg', '.html', '.xml', '.txt', '.webmanifest']);
const PROSE = new Set(['.md', '.mdx', '.yaml', '.yml']);
const PDF = 'dist/blake-bailey-resume.pdf';

/** @param {string} dir @returns {AsyncGenerator<string>} */
async function* walk(dir) {
  for (const entry of await readdir(dir, { withFileTypes: true })) {
    const path = join(dir, entry.name);
    if (entry.isDirectory()) yield* walk(path);
    else yield path;
  }
}

if (!existsSync('dist') || !existsSync(PDF)) {
  console.error('Run `npm run build` first: the check reads dist/ and the resume PDF.');
  process.exit(1);
}

/** @type {string[]} */
const problems = [];
/** @param {string} where @param {{ rule: string, message: string, line: number }[]} found */
const report = (where, found) => {
  for (const v of found) problems.push(`${where}:${v.line}: ${v.message} [${v.rule}]`);
};

let files = 0;
for (const dir of ['src', 'public']) {
  for await (const file of walk(dir)) {
    const ext = extname(file);
    if (!TEXT.has(ext)) continue;
    files++;
    report(file, findViolations(await readFile(file, 'utf8'), PROSE.has(ext) ? 'all' : 'general'));
  }
}
for await (const file of walk('dist')) {
  const ext = extname(file);
  if (!TEXT.has(ext)) continue;
  files++;
  const raw = await readFile(file, 'utf8');
  report(file, findViolations(raw, 'general'));
  if (ext === '.html') report(`${file} (visible text)`, findViolations(visibleText(raw), 'prose'));
}
report(`${PDF} (text)`, findViolations((await pdfText(PDF)).text, 'all'));

if (problems.length > 0) {
  console.error(`Content check failed:\n${problems.join('\n')}`);
  process.exit(1);
}
console.log(`Content check passed: ${files} files and the resume PDF.`);
```
